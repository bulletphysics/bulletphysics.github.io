void Ball::throw(const Vector3 force)
{	
	// Lock the ball --> player can't control it anymore
	locked = true;

	this->mRigidBody.activate(true);

	this->mRigidBody.applyCentralImpulse(btVector3(force.x,force.y,force.z));	

}
