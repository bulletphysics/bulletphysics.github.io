        self.sizer.Add(self.panel, 0, wx.EXPAND, 0)
        self.SetSizer(self.sizer)


class Panel(wx.Panel):
    def __init__(self, parent):
        wx.Panel.__init__(self, parent = parent)

    def start(self, event):
       sim.start

    frame = MainFrame()
    frame.Show()
    app.MainLoop()
