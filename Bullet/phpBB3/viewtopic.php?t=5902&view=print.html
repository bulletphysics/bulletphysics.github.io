Code: Select all

$selected = `ls -sl`;

for($node in $selected)
{
    select -r $node;
    pickWalk -d down;
    $temp = `ls -sl`;
    setAttr ($temp[0] + ".visibility") 0;
    $temp = `listConnections`;
    select -r $temp[0];
    $temp = `listConnections`;
    select -r $temp[1];
    showHidden -a;
    select -add $node;
    parent;
}
